# mingol_add_test(<name> SOURCES <files...> LIBS <targets...>)
function(mingol_add_test name)
  cmake_parse_arguments(T "" "" "SOURCES;LIBS" ${ARGN})
  add_executable(${name} ${T_SOURCES})
  target_link_libraries(${name} PRIVATE mingol_test_main ${T_LIBS})
  mingol_target_options(${name})
  add_test(NAME ${name} COMMAND ${name})
  if(MINGOL_SANITIZE AND NOT MSVC)
    # LeakSanitizer is off here: the Vulkan loader dlclose()s the driver, which leaves leak stacks
    # unsymbolised and unsuppressible. Address/UB checks stay on; leaks are covered by the valgrind
    # test below (MINGOL_VALGRIND_TESTS).
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  endif()
  if(MINGOL_TSAN AND NOT MSVC)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT
      "TSAN_OPTIONS=suppressions=${PROJECT_SOURCE_DIR}/tests/tsan.supp:halt_on_error=0:exitcode=66")
  endif()
endfunction()

option(MINGOL_VALGRIND_TESTS "Also register valgrind leak checks (slow, needs valgrind)" OFF)
